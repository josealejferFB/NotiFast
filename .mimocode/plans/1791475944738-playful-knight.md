# Note Editor Premium — Bottom Sheet + Rich Text + Tags Implementation Plan

> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** Convert the note editor into a unified expandable bottom sheet (create + edit share one entry point) with rich-text formatting spans, authorable thematic tags, editable lists (ordered + unordered), and fix the critical/high bugs that make the current editor incomplete.

**Architecture:** Single `NoteEditorBottomSheet` composable registered as a Navigation Compose `dialog()` destination. Existing `NoteEditorViewModel` + `NoteBlock` model + `NoteBlockCodec` are reused; we add UI layers that wire dormant VM events (`TagToggled`, `BlockSpansChanged`, `CheckItemAdded/Removed`) and a `RichBlockTextField` that edits `TextSpan` styles over text selection. `BulletList` gains an `isOrdered` flag instead of a new sealed subtype.

**Tech Stack:** Kotlin 2.0.21, Jetpack Compose Material 3, Hilt, Room (offline-first), Coil, Navigation Compose 2.8.5.

**Spec:** This plan (design decisions inline). Diagnostic evidence from systematic-debugging exploration of `presentation/feature/noteeditor/` and `presentation/feature/notes/`.

```kotlin
package com.notifast.app.domain.util

import com.notifast.app.data.model.TextSpan
import com.notifast.app.data.model.TextStyle

/**
 * Pure span algebra over [TextSpan] ranges. Ranges are [start, end) half-open.
 */
object TextSpanOps {

    fun toggleStyle(
        spans: List<TextSpan>,
        start: Int,
        end: Int,
        style: TextStyle
    ): List<TextSpan> {
        if (start >= end) return spans
        val covering = spans.filter { it.start < end && it.end > start && style in it.styles }
        val next = if (covering.isEmpty()) {
            addStyle(spans, start, end, style)
        } else {
            removeStyle(spans, start, end, style)
        }
        return normalize(next)
    }

    private fun addStyle(
        spans: List<TextSpan>,
        start: Int,
        end: Int,
        style: TextStyle
    ): List<TextSpan> {
        val others = subtractRange(spans, start, end, onlyStyle = null)
        return others + TextSpan(start, end, setOf(style))
    }

    private fun removeStyle(
        spans: List<TextSpan>,
        start: Int,
        end: Int,
        style: TextStyle
    ): List<TextSpan> {
        val result = mutableListOf<TextSpan>()
        for (span in spans) {
            if (span.start >= end || span.end <= start || style !in span.styles) {
                result += span
                continue
            }
            if (span.start < start) {
                result += span.copy(end = start)
            }
            if (span.end > end) {
                result += span.copy(start = end)
            }
            val remaining = span.styles - style
            if (remaining.isNotEmpty() && span.start < end && span.end > start) {
                val midStart = maxOf(span.start, start)
                val midEnd = minOf(span.end, end)
                if (midStart < midEnd) result += TextSpan(midStart, midEnd, remaining)
            }
        }
        return result
    }

    private fun subtractRange(
        spans: List<TextSpan>,
        start: Int,
        end: Int,
        onlyStyle: TextStyle?
    ): List<TextSpan> {
        val result = mutableListOf<TextSpan>()
        for (span in spans) {
            if (span.start >= end || span.end <= start) {
                result += span
                continue
            }
            val styles = if (onlyStyle == null) emptySet() else span.styles - onlyStyle
            if (span.start < start) {
                result += span.copy(end = start)
            }
            if (span.end > end) {
                result += span.copy(start = end)
            }
            if (styles.isNotEmpty()) {
                val midStart = maxOf(span.start, start)
                val midEnd = minOf(span.end, end)
                if (midStart < midEnd) result += TextSpan(midStart, midEnd, styles)
            }
        }
        return result
    }

    fun normalize(spans: List<TextSpan>): List<TextSpan> {
        val sorted = spans
            .filter { it.start < it.end && it.styles.isNotEmpty() }
            .sortedWith(compareBy({ it.start }, { it.end }))
        if (sorted.isEmpty()) return emptyList()
        val merged = mutableListOf<TextSpan>()
        var current = sorted.first()
        for (i in 1 until sorted.size) {
            val next = sorted[i]
            if (current.end == next.start && current.styles == next.styles) {
                current = current.copy(end = next.end)
            } else {
                merged += current
                current = next
            }
        }
        merged += current
        return merged
    }

    /**
     * R1 mitigation: re-map span boundaries after a text edit.
     * [editStart] is the caret/anchor where the edit occurred.
     * [oldLength] / [newLength] are the replaced and inserted lengths (0 for pure insert/delete).
     * Ranges are half-open [start, end).
     */
    fun adjustForEdit(
        spans: List<TextSpan>,
        editStart: Int,
        oldLength: Int,
        newLength: Int
    ): List<TextSpan> {
        if (spans.isEmpty()) return emptyList()
        val delta = newLength - oldLength
        val editEndOld = editStart + oldLength
        val adjusted = spans.mapNotNull { span ->
            var start = span.start
            var end = span.end
            if (span.start >= editEndOld) {
                start += delta
                end += delta
            } else if (span.end <= editStart) {
                // entirely before edit — unchanged
            } else {
                // overlaps the edit region
                if (start > editStart) {
                    start = editStart + newLength
                }
                if (end >= editEndOld) {
                    end += delta
                } else {
                    end = editStart + newLength
                }
                if (start > editStart && start > editEndOld) start = editStart + newLength
                // clamp: span collapsed?
                if (start > end) {
                    start = editStart
                    end = editStart + newLength
                }
                // if deletion removed the whole span content
                if (oldLength > 0 && newLength == 0 && span.start >= editStart && span.end <= editEndOld) {
                    return@mapNotNull null
                }
            }
            if (end <= start) null else span.copy(start = start, end = end)
        }
        return normalize(adjusted)
    }

    /** Styles that cover the entire [start, end) range (used by the format toolbar). */
    fun stylesAt(spans: List<TextSpan>, start: Int, end: Int): Set<TextStyle> {
        if (start >= end) return emptySet()
        val covering = spans.filter { it.start <= start && it.end >= end }
        if (covering.isEmpty()) return emptySet()
        return covering.map { it.styles }.reduce { acc, s -> acc intersect s }
    }
}
```

```kotlin
// R4: unique name — does NOT shadow androidx.compose.ui.text.buildAnnotatedString
fun applyNoteSpans(text: String, spans: List<TextSpan>): AnnotatedString =
    buildAnnotatedString {
        append(text)
        spans.forEach { span ->
            val start = span.start.coerceIn(0, text.length)
            val end = span.end.coerceIn(0, text.length)
            if (start >= end) return@forEach
            span.styles.forEach { style ->
                addStyle(style.toSpanStyle(), start, end)
            }
        }
    }

private fun NoteTextStyle.toSpanStyle(): SpanStyle = when (this) {
    NoteTextStyle.BOLD -> SpanStyle(fontWeight = FontWeight.Bold)
    NoteTextStyle.ITALIC -> SpanStyle(fontStyle = FontStyle.Italic)
    NoteTextStyle.UNDERLINE -> SpanStyle(textDecoration = TextDecoration.Underline)
    NoteTextStyle.STRIKETHROUGH -> SpanStyle(textDecoration = TextDecoration.LineThrough)
    NoteTextStyle.HIGHLIGHT -> SpanStyle(background = HighlightWash)
}

@Composable
fun RichBlockTextField(
    value: String,
    spans: List<TextSpan>,
    onValueChange: (String) -> Unit,
    onSpansChange: (List<TextSpan>) -> Unit,
    onFocus: () -> Unit,
    onSelectionChange: (Int, Int) -> Unit,
    textStyle: TextStyle,
    isFocused: Boolean,
    modifier: Modifier = Modifier,
    placeholder: String = ""
) {
    // R2 mitigation: TextFieldValue state WITHOUT key on `value`.
    // We only resync when an EXTERNAL change arrives (parent set a different text
    // than what we currently hold) — never on our own onValueChange cycle.
    var fieldValue by remember { mutableStateOf(TextFieldValue(text = value)) }
    if (value != fieldValue.text) {
        // External reset (block switch, load, undo) — adopt parent text, caret at end
        fieldValue = TextFieldValue(text = value, selection = TextRange(value.length))
    }
    val annotated = remember(fieldValue.text, spans) { applyNoteSpans(fieldValue.text, spans) }

    BasicTextField(
        value = fieldValue.copy(annotatedString = annotated),
        onValueChange = { new ->
            val oldText = fieldValue.text
            val newText = new.text
            if (newText != oldText) {
                // R1 mitigation: re-map span indices before emitting
                val editStart = findEditStart(oldText, newText)
                val oldLen = oldText.length - editStart - commonSuffix(oldText, newText, editStart)
                val newLen = newText.length - editStart - commonSuffix(oldText, newText, editStart)
                val adjusted = TextSpanOps.adjustForEdit(spans, editStart, oldLen, newLen)
                fieldValue = new
                onValueChange(newText)
                onSpansChange(adjusted)
            } else {
                fieldValue = new
            }
            onSelectionChange(new.selection.min, new.selection.max)
        },
        textStyle = textStyle.copy(color = MaterialTheme.colorScheme.onSurface),
        cursorBrush = SolidColor(MaterialTheme.colorScheme.primary),
        modifier = modifier.onFocusChanged { if (it.isFocused) onFocus() },
        decorationBox = { inner ->
            if (value.isEmpty() && placeholder.isNotEmpty()) {
                Text(
                    placeholder,
                    style = textStyle.copy(
                        color = MaterialTheme.colorScheme.onSurfaceVariant.copy(alpha = 0.5f)
                    )
                )
            }
            inner()
        }
    )
}

private fun commonSuffix(a: String, b: String, editStart: Int): Int {
    var i = 0
    while (i < a.length - editStart && i < b.length - editStart &&
        a[a.length - 1 - i] == b[b.length - 1 - i]
    ) i++
    return i
}

private fun findEditStart(a: String, b: String): Int {
    val max = minOf(a.length, b.length)
    var i = 0
    while (i < max && a[i] == b[i]) i++
    return i
}
```
