```kotlin
package com.notifast.app.data.notification

import android.app.NotificationChannel
import android.app.NotificationManager
import android.content.Context
import android.os.Build
import androidx.core.app.NotificationManagerCompat
import com.notifast.app.R

object NotificationChannelsManager {
    const val CHANNEL_EVALUATIONS = "notifast_evaluations"
    const val CHANNEL_CLASSES = "notifast_classes"
    const val CHANNEL_DAILY_SUMMARY = "notifast_daily_summary"

    fun createChannels(context: Context) {
        if (Build.VERSION.SDK_INT >= Build.VERSION_CODES.O) {
            val evalChannel = NotificationChannel(
                CHANNEL_EVALUATIONS,
                context.getString(R.string.channel_evaluations_name),
                NotificationManager.IMPORTANCE_HIGH
            ).apply {
                description = context.getString(R.string.channel_evaluations_desc)
                enableVibration(true)
            }

            val classChannel = NotificationChannel(
                CHANNEL_CLASSES,
                context.getString(R.string.channel_classes_name),
                NotificationManager.IMPORTANCE_HIGH
            ).apply {
                description = context.getString(R.string.channel_classes_desc)
                enableVibration(true)
            }

            val summaryChannel = NotificationChannel(
                CHANNEL_DAILY_SUMMARY,
                context.getString(R.string.channel_daily_summary_name),
                NotificationManager.IMPORTANCE_DEFAULT
            ).apply {
                description = context.getString(R.string.channel_daily_summary_desc)
            }

            val manager = context.getSystemService(Context.NOTIFICATION_SERVICE) as? NotificationManager
            manager?.createNotificationChannels(listOf(evalChannel, classChannel, summaryChannel))
        }
    }
}
```

```kotlin
package com.notifast.app.data.notification

import android.Manifest
import android.app.PendingIntent
import android.content.Context
import android.content.Intent
import android.content.pm.PackageManager
import android.os.Build
import androidx.core.app.ActivityCompat
import androidx.core.app.NotificationCompat
import androidx.core.app.NotificationManagerCompat
import com.notifast.app.MainActivity
import com.notifast.app.R
import dagger.hilt.android.qualifiers.ApplicationContext
import javax.inject.Inject
import javax.inject.Singleton

@Singleton
class NotificationSender @Inject constructor(
    @ApplicationContext private val context: Context
) {

    fun sendEvaluationNotification(
        notificationId: Int,
        subjectName: String,
        evalTitle: String,
        dateText: String,
        weight: Double
    ) {
        if (!hasNotificationPermission()) return

        val intent = Intent(context, MainActivity::class.java).apply {
            flags = Intent.FLAG_ACTIVITY_NEW_TASK or Intent.FLAG_ACTIVITY_CLEAR_TASK
        }
        val pendingIntent = PendingIntent.getActivity(
            context,
            notificationId,
            intent,
            PendingIntent.FLAG_UPDATE_CURRENT or PendingIntent.FLAG_IMMUTABLE
        )

        val body = context.getString(
            R.string.notification_eval_body,
            subjectName,
            dateText,
            String.format(java.util.Locale.US, "%.0f", weight)
        )

        val notification = NotificationCompat.Builder(context, NotificationChannelsManager.CHANNEL_EVALUATIONS)
            .setSmallIcon(R.drawable.logo)
            .setContentTitle(context.getString(R.string.notification_eval_title, evalTitle))
            .setContentText(body)
            .setStyle(NotificationCompat.BigTextStyle().bigText(body))
            .setPriority(NotificationCompat.PRIORITY_HIGH)
            .setContentIntent(pendingIntent)
            .setAutoCancel(true)
            .build()

        NotificationManagerCompat.from(context).notify(notificationId, notification)
    }

    fun sendClassNotification(
        notificationId: Int,
        subjectName: String,
        classroom: String,
        startTime: String
    ) {
        if (!hasNotificationPermission()) return

        val intent = Intent(context, MainActivity::class.java).apply {
            flags = Intent.FLAG_ACTIVITY_NEW_TASK or Intent.FLAG_ACTIVITY_CLEAR_TASK
        }
        val pendingIntent = PendingIntent.getActivity(
            context,
            notificationId,
            intent,
            PendingIntent.FLAG_UPDATE_CURRENT or PendingIntent.FLAG_IMMUTABLE
        )

        val body = context.getString(R.string.notification_class_body, startTime, classroom.ifBlank { "N/A" })

        val notification = NotificationCompat.Builder(context, NotificationChannelsManager.CHANNEL_CLASSES)
            .setSmallIcon(R.drawable.logo)
            .setContentTitle(context.getString(R.string.notification_class_title, subjectName))
            .setContentText(body)
            .setStyle(NotificationCompat.BigTextStyle().bigText(body))
            .setPriority(NotificationCompat.PRIORITY_HIGH)
            .setContentIntent(pendingIntent)
            .setAutoCancel(true)
            .build()

        NotificationManagerCompat.from(context).notify(notificationId, notification)
    }

    fun sendDailySummaryNotification(
        notificationId: Int,
        summaryBody: String
    ) {
        if (!hasNotificationPermission()) return

        val intent = Intent(context, MainActivity::class.java).apply {
            flags = Intent.FLAG_ACTIVITY_NEW_TASK or Intent.FLAG_ACTIVITY_CLEAR_TASK
        }
        val pendingIntent = PendingIntent.getActivity(
            context,
            notificationId,
            intent,
            PendingIntent.FLAG_UPDATE_CURRENT or PendingIntent.FLAG_IMMUTABLE
        )

        val notification = NotificationCompat.Builder(context, NotificationChannelsManager.CHANNEL_DAILY_SUMMARY)
            .setSmallIcon(R.drawable.logo)
            .setContentTitle(context.getString(R.string.notification_daily_summary_title))
            .setContentText(summaryBody)
            .setStyle(NotificationCompat.BigTextStyle().bigText(summaryBody))
            .setPriority(NotificationCompat.PRIORITY_DEFAULT)
            .setContentIntent(pendingIntent)
            .setAutoCancel(true)
            .build()

        NotificationManagerCompat.from(context).notify(notificationId, notification)
    }

    private fun hasNotificationPermission(): Boolean {
        return if (Build.VERSION.SDK_INT >= Build.VERSION_CODES.TIRAMISU) {
            ActivityCompat.checkSelfPermission(
                context,
                Manifest.permission.POST_NOTIFICATIONS
            ) == PackageManager.PERMISSION_GRANTED
        } else {
            true
        }
    }
}
```

```kotlin
package com.notifast.app.data.worker

import android.content.Context
import androidx.hilt.work.HiltWorker
import androidx.work.CoroutineWorker
import androidx.work.WorkerParameters
import com.notifast.app.data.notification.NotificationSender
import com.notifast.app.domain.repository.AuthRepository
import com.notifast.app.domain.repository.EvaluationRepository
import com.notifast.app.domain.repository.UserPrefsRepository
import dagger.assisted.Assisted
import dagger.assisted.AssistedInject
import kotlinx.coroutines.flow.first

@HiltWorker
class EvaluationReminderWorker @AssistedInject constructor(
    @Assisted appContext: Context,
    @Assisted workerParams: WorkerParameters,
    private val notificationSender: NotificationSender,
    private val evaluationRepository: EvaluationRepository,
    private val authRepository: AuthRepository,
    private val userPrefsRepository: UserPrefsRepository
) : CoroutineWorker(appContext, workerParams) {

    override suspend fun doWork(): Result {
        val evalAlertsEnabled = userPrefsRepository.evalAlertsEnabledFlow.first()
        if (!evalAlertsEnabled) return Result.success()

        val evaluationId = inputData.getString(KEY_EVALUATION_ID) ?: return Result.failure()
        val subjectName = inputData.getString(KEY_SUBJECT_NAME) ?: ""
        val title = inputData.getString(KEY_TITLE) ?: ""
        val dateText = inputData.getString(KEY_DATE) ?: ""
        val weight = inputData.getDouble(KEY_WEIGHT, 0.0)

        val userId = authRepository.getCurrentUserId()
        if (userId != null) {
            val eval = evaluationRepository.getEvaluation(userId, evaluationId).first()
            if (eval == null || eval.grade != null) {
                return Result.success()
            }
        }

        notificationSender.sendEvaluationNotification(
            notificationId = evaluationId.hashCode(),
            subjectName = subjectName,
            evalTitle = title,
            dateText = dateText,
            weight = weight
        )

        return Result.success()
    }

    companion object {
        const val KEY_EVALUATION_ID = "eval_id"
        const val KEY_SUBJECT_NAME = "subject_name"
        const val KEY_TITLE = "title"
        const val KEY_DATE = "date"
        const val KEY_WEIGHT = "weight"
    }
}
```

```kotlin
package com.notifast.app.domain.usecase

import androidx.work.Data
import androidx.work.ExistingWorkPolicy
import androidx.work.OneTimeWorkRequestBuilder
import androidx.work.WorkManager
import com.notifast.app.data.model.Evaluation
import com.notifast.app.data.worker.EvaluationReminderWorker
import com.notifast.app.domain.repository.UserPrefsRepository
import kotlinx.coroutines.flow.first
import java.text.SimpleDateFormat
import java.util.Calendar
import java.util.Date
import java.util.Locale
import java.util.concurrent.TimeUnit
import javax.inject.Inject

class ScheduleEvaluationAlertUseCase @Inject constructor(
    private val workManager: WorkManager,
    private val userPrefsRepository: UserPrefsRepository
) {
    private val dateFormat = SimpleDateFormat("yyyy-MM-dd", Locale.getDefault())

    suspend operator fun invoke(evaluation: Evaluation, customDaysInAdvance: Int? = null) {
        val enabled = userPrefsRepository.evalAlertsEnabledFlow.first()
        if (!enabled) return

        val daysInAdvance = customDaysInAdvance ?: userPrefsRepository.evalAlertDaysFlow.first()
        val evalDate: Date = try {
            dateFormat.parse(evaluation.date) ?: return
        } catch (e: Exception) {
            return
        }

        val targetCal = Calendar.getInstance().apply {
            time = evalDate
            add(Calendar.DAY_OF_YEAR, -daysInAdvance)
            set(Calendar.HOUR_OF_DAY, 9)
            set(Calendar.MINUTE, 0)
            set(Calendar.SECOND, 0)
            set(Calendar.MILLISECOND, 0)
        }

        val now = System.currentTimeMillis()
        var delayMillis = targetCal.timeInMillis - now

        if (delayMillis < 0) {
            // Si la hora de 9 AM ya pasó, pero el examen es hoy/futuro, alertar en 5 minutos
            val examCal = Calendar.getInstance().apply { time = evalDate; set(Calendar.HOUR_OF_DAY, 23); set(Calendar.MINUTE, 59) }
            if (examCal.timeInMillis > now) {
                delayMillis = TimeUnit.MINUTES.toMillis(5)
            } else {
                return
            }
        }

        val inputData = Data.Builder()
            .putString(EvaluationReminderWorker.KEY_EVALUATION_ID, evaluation.id)
            .putString(EvaluationReminderWorker.KEY_SUBJECT_NAME, evaluation.subjectName)
            .putString(EvaluationReminderWorker.KEY_TITLE, evaluation.title)
            .putString(EvaluationReminderWorker.KEY_DATE, evaluation.date)
            .putDouble(EvaluationReminderWorker.KEY_WEIGHT, evaluation.weight)
            .build()

        val workRequest = OneTimeWorkRequestBuilder<EvaluationReminderWorker>()
            .setInitialDelay(delayMillis, TimeUnit.MILLISECONDS)
            .setInputData(inputData)
            .addTag("eval_${evaluation.id}")
            .build()

        workManager.enqueueUniqueWork(
            "eval_${evaluation.id}",
            ExistingWorkPolicy.REPLACE,
            workRequest
        )
    }
}
```

```kotlin
package com.notifast.app.domain.usecase

import androidx.work.Data
import androidx.work.ExistingWorkPolicy
import androidx.work.OneTimeWorkRequestBuilder
import androidx.work.WorkManager
import com.notifast.app.data.model.Schedule
import com.notifast.app.data.worker.ClassScheduleWorker
import com.notifast.app.domain.repository.UserPrefsRepository
import kotlinx.coroutines.flow.first
import java.util.Calendar
import java.util.concurrent.TimeUnit
import javax.inject.Inject

class ScheduleClassAlertUseCase @Inject constructor(
    private val workManager: WorkManager,
    private val userPrefsRepository: UserPrefsRepository
) {
    suspend operator fun invoke(schedule: Schedule, customMinutesInAdvance: Int? = null) {
        val enabled = userPrefsRepository.classAlertsEnabledFlow.first()
        if (!enabled) return

        val minutesInAdvance = customMinutesInAdvance ?: userPrefsRepository.classAlertMinutesFlow.first()
        val parts = schedule.startTime.split(":")
        if (parts.size != 2) return
        val hour = parts[0].toIntOrNull() ?: return
        val minute = parts[1].toIntOrNull() ?: return

        val targetCal = Calendar.getInstance().apply {
            val currentDay = (get(Calendar.DAY_OF_WEEK) - 1).let { if (it == 0) 0 else it }
            var daysUntilTarget = (schedule.dayOfWeek - currentDay + 7) % 7
            
            set(Calendar.HOUR_OF_DAY, hour)
            set(Calendar.MINUTE, minute)
            set(Calendar.SECOND, 0)
            set(Calendar.MILLISECOND, 0)
            add(Calendar.MINUTE, -minutesInAdvance)

            if (daysUntilTarget == 0 && timeInMillis <= System.currentTimeMillis()) {
                daysUntilTarget = 7
            }
            if (daysUntilTarget > 0) {
                add(Calendar.DAY_OF_YEAR, daysUntilTarget)
            }
        }

        val delayMillis = targetCal.timeInMillis - System.currentTimeMillis()
        if (delayMillis < 0) return

        val inputData = Data.Builder()
            .putString(ClassScheduleWorker.KEY_SCHEDULE_ID, schedule.id)
            .putString(ClassScheduleWorker.KEY_SUBJECT_NAME, schedule.subjectName)
            .putString(ClassScheduleWorker.KEY_START_TIME, schedule.startTime)
            .putString(ClassScheduleWorker.KEY_CLASSROOM, schedule.classroom)
            .build()

        val workRequest = OneTimeWorkRequestBuilder<ClassScheduleWorker>()
            .setInitialDelay(delayMillis, TimeUnit.MILLISECONDS)
            .setInputData(inputData)
            .addTag("class_${schedule.id}")
            .build()

        workManager.enqueueUniqueWork(
            "class_${schedule.id}",
            ExistingWorkPolicy.REPLACE,
            workRequest
        )
    }
}
```

```kotlin
class DailySummaryWorker @AssistedInject constructor(
    @Assisted appContext: Context,
    @Assisted workerParams: WorkerParameters,
    private val notificationSender: NotificationSender,
    private val scheduleRepository: ScheduleRepository,
    private val evaluationRepository: EvaluationRepository,
    private val authRepository: AuthRepository,
    private val userPrefsRepository: UserPrefsRepository

    private val dateFormat = SimpleDateFormat("yyyy-MM-dd", Locale.getDefault())

    override suspend fun doWork(): Result {
        val dailySummaryEnabled = userPrefsRepository.dailySummaryEnabledFlow.first()
        if (!dailySummaryEnabled) return Result.success()

        val userId = authRepository.getCurrentUserId() ?: return Result.success()

        val cal = Calendar.getInstance()
        val todayDow = (cal.get(Calendar.DAY_OF_WEEK) - 1).let { if (it == 0) 0 else it }
        val schedules = scheduleRepository.getSchedules(userId).first().filter { it.dayOfWeek == todayDow }

        val todayDate = dateFormat.format(cal.time)
        cal.add(Calendar.DAY_OF_YEAR, 2)
        val futureDate = dateFormat.format(cal.time)

        val pendingEvals = evaluationRepository.getEvaluations(userId).first().filter {
            it.date in todayDate..futureDate && it.grade == null
        }

        if (schedules.isEmpty() && pendingEvals.isEmpty()) return Result.success()

        val classesSummary = if (schedules.isNotEmpty()) "${schedules.size} clase(s) hoy" else "Sin clases hoy"
        val evalsSummary = if (pendingEvals.isNotEmpty()) "${pendingEvals.size} evaluación(es) en 48h" else "Sin entregas inminentes"
        val body = "$classesSummary • $evalsSummary"

        notificationSender.sendDailySummaryNotification(
            notificationId = 1001,
            summaryBody = body
        )

        return Result.success()
    }
```

```kotlin
package com.notifast.app.domain.usecase

import androidx.work.ExistingPeriodicWorkPolicy
import androidx.work.PeriodicWorkRequestBuilder
import androidx.work.WorkManager
import com.notifast.app.data.worker.DailySummaryWorker
import com.notifast.app.domain.repository.AuthRepository
import com.notifast.app.domain.repository.EvaluationRepository
import com.notifast.app.domain.repository.ScheduleRepository
import com.notifast.app.domain.repository.UserPrefsRepository
import kotlinx.coroutines.flow.first
import java.util.Calendar
import java.util.concurrent.TimeUnit
import javax.inject.Inject

class RescheduleAllAlertsUseCase @Inject constructor(
    private val workManager: WorkManager,
    private val authRepository: AuthRepository,
    private val evaluationRepository: EvaluationRepository,
    private val scheduleRepository: ScheduleRepository,
    private val userPrefsRepository: UserPrefsRepository,
    private val scheduleEvaluationAlertUseCase: ScheduleEvaluationAlertUseCase,
    private val scheduleClassAlertUseCase: ScheduleClassAlertUseCase
) {
    suspend operator fun invoke() {
        val userId = authRepository.getCurrentUserId() ?: return

        // 1. Sincronizar evaluaciones activas
        val evaluations = evaluationRepository.getEvaluations(userId).first()
        evaluations.filter { it.grade == null }.forEach { eval ->
            scheduleEvaluationAlertUseCase(eval)
        }

        // 2. Sincronizar horarios de clases
        val schedules = scheduleRepository.getSchedules(userId).first()
        schedules.forEach { sch ->
            scheduleClassAlertUseCase(sch)
        }

        // 3. Resumen diario periódico si está activo
        val dailySummaryEnabled = userPrefsRepository.dailySummaryEnabledFlow.first()
        if (dailySummaryEnabled) {
            val initialDelay = calculateDailySummaryInitialDelay()
            val summaryRequest = PeriodicWorkRequestBuilder<DailySummaryWorker>(1, TimeUnit.DAYS)
                .setInitialDelay(initialDelay, TimeUnit.MILLISECONDS)
                .addTag("daily_summary")
                .build()

            workManager.enqueueUniquePeriodicWork(
                "daily_summary",
                ExistingPeriodicWorkPolicy.UPDATE,
                summaryRequest
            )
        } else {
            workManager.cancelUniqueWork("daily_summary")
        }
    }

    private fun calculateDailySummaryInitialDelay(): Long {
        val now = Calendar.getInstance()
        val target = Calendar.getInstance().apply {
            set(Calendar.HOUR_OF_DAY, 7)
            set(Calendar.MINUTE, 0)
            set(Calendar.SECOND, 0)
            set(Calendar.MILLISECOND, 0)
            if (before(now)) {
                add(Calendar.DAY_OF_YEAR, 1)
            }
        }
        return target.timeInMillis - now.timeInMillis
    }
}
```
